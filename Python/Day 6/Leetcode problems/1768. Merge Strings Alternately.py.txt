class Solution:
    def mergeAlternately(self, word1: str, word2: str) -> str:
        my_list=[len(word1),len(word2)]
        result=[]
        max_len=max(my_list)
        for i in range(max_len):
            if i<len(word1):
                result.append(word1[i])
            if i<len(word2):
                result.append(word2[i])
        str_result="".join(result)
        return str_result
